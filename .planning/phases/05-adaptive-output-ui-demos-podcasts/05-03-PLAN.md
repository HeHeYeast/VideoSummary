### UI demo 4 子规则 (D-20 字面 verbatim)

skeleton 必须按以下 D-20 字面顺序嵌入 CLAUDE.md：

1. **Pixel-text 不确定引用** — UI 软件用 proportional 字体 + 抗锯齿，多模态识别准度低于 monospace；指令 Claude `quote-with-uncertainty`，用 `> 该控件大致写着"XXX"（多帧交叉验证后置信度中等）`
2. **Tooltip 遮挡检测** — 发现 tooltip 遮目标 → 采前/后 0.5s 帧；都遮 → 标 `*该值视图被 tooltip 遮挡，未取得*`
3. **光标不可见兜底** — 黑光标在黑 UI / 截屏不含光标 → 从前后帧 panel state diff 推点击位置；命名一律按控件 label/icon，禁空间方位
4. **`--width 1280/1920` 4K override** — 4K 录屏抽 frame 时若 `default_scale: 854:-1` 输出过小（< 800px 宽）丢可读性 → schedule.json 加 `default_scale: 1280:-1` 或 `1920:-1` 提示

```python
"""Speaker diarization via pyannote.audio (opt-in via requirements-optional.txt).

Phase 5 TEACH-08 / D-13..D-17.

Lazy-imports pyannote.audio so missing dep raises a clean RuntimeError with
install hint (matches Phase 4 detect_silence opt-in idiom).

NOTE: pyannote 4.0 model `pyannote/speaker-diarization-community-1` requires:
  1. user accept license at HF model card
  2. HF_TOKEN env var (read from .env via load_dotenv())

Output schema (D-13):
  [{"start": <float>, "end": <float>, "speaker_id": "SPEAKER_NN"}, ...]
"""
from __future__ import annotations

import logging
import os
from pathlib import Path

log = logging.getLogger(__name__)


def diarize_audio(
    audio_path: str | Path,
    *,
    hf_token: str,
    allow_long: bool = False,
) -> list[dict]:
    """Run pyannote speaker diarization on a wav file.
    
    Args:
        audio_path: path to audio.wav (16kHz mono PCM preferred)
        hf_token: HuggingFace API token (read by caller from .env HF_TOKEN)
        allow_long: passthrough flag (caller should have already gated on duration)
    
    Returns:
        list of {"start": float, "end": float, "speaker_id": str} dicts
    
    Raises:
        RuntimeError: pyannote.audio not installed
        RuntimeError: HF model load failed (license not accepted / token invalid)
    """
    try:
        # Lazy import — keep agent/diarize.py importable even when pyannote not installed
        from pyannote.audio import Pipeline
    except ImportError as e:
        raise RuntimeError(
            "pyannote.audio not installed; install via "
            "`pip install -r requirements-optional.txt` (~700MB torch dep)"
        ) from e
    
    audio_path = Path(audio_path)
    if not audio_path.exists():
        raise RuntimeError(f"audio file not found: {audio_path}")
    
    # NOTE: do NOT log hf_token (T-05-03-08); only log presence
    log.info("loading pyannote/speaker-diarization-community-1 (HF_TOKEN: %s)",
             "<set>" if hf_token else "<empty>")
    
    try:
        pipeline = Pipeline.from_pretrained(
            "pyannote/speaker-diarization-community-1",
            use_auth_token=hf_token,
        )
    except Exception as e:
        raise RuntimeError(
            f"failed to load pyannote model; verify HF token + community-1 license accepted "
            f"at https://huggingface.co/pyannote/speaker-diarization-community-1: {e}"
        ) from e
    
    log.info("running diarization on %s", audio_path)
    diarization = pipeline(str(audio_path))
    
    turns = []
    for turn, _, speaker in diarization.itertracks(yield_label=True):
        turns.append({
            "start": float(turn.start),
            "end": float(turn.end),
            "speaker_id": str(speaker),
        })
    log.info("diarization complete: %d turns", len(turns))
    return turns
```

def cmd_diarize(args):
    """Speaker diarization (opt-in via pyannote.audio).
    
    Phase 5 TEACH-08 / D-13..D-17. Emits output/<slug>/diarization.json.
    """
    sys.path.insert(0, str(Path(__file__).parent.parent))
    from agent.diarize import diarize_audio
    from agent.sources._common import ffprobe_video
    
    out_path = Path(args.out)
    out_dir = out_path.parent
    out_dir.mkdir(parents=True, exist_ok=True)
    
    # 1. HF_TOKEN guard (CONTEXT D-15)
    hf_token = os.environ.get("HF_TOKEN", "").strip()
    if not hf_token:
        raise RuntimeError(
            "HF_TOKEN not set; see CLAUDE.md '## Pyannote diarization 设置'"
        )
    
    # 2. ffprobe duration probe + 60min+ CPU warning gate (D-16 字面 verbatim)
    audio_path = Path(args.audio_wav)
    if not audio_path.exists():
        raise RuntimeError(f"audio file not found: {audio_path}")
    
    duration_s = 0
    try:
        info = ffprobe_video(audio_path)
        duration_s = info.get("duration_s", 0) or 0
    except Exception as e:
        log.warning("ffprobe failed on %s: %s; skipping duration gate", audio_path, e)
    
    # CUDA detect (lazy)
    has_cuda = False
    try:
        import torch
        has_cuda = torch.cuda.is_available()
    except ImportError:
        pass
    
    if duration_s > 3600 and not has_cuda and not args.allow_long:
        # D-16 字面 prompt verbatim
        print(
            f"WARNING: 60min+ 音频在 CPU 上 pyannote 预计 3-5× wall time"
            f"（约 3-5h）。建议 (1) 切片处理 (2) 跳过 diarization 让 Claude "
            f"从内容推断 (3) 等待 GPU 机器再跑。继续？(y/N)"
        )
        ans = input().strip().lower()
        if ans != "y":
            raise RuntimeError("diarize aborted by user (60min+ CPU gate)")
    
    # 3. Phase 2 sidecar idiom — diarize 也走 sidecar/state.jsonl
    state_dir = out_dir
    current_sidecar = _build_sidecar(
        cli={"audio_wav": str(audio_path)},
        func={
            "model": "pyannote/speaker-diarization-community-1",
            "device": "cuda" if has_cuda else "cpu",
            "duration_s": duration_s,
        },
        tools={},  # pyannote.audio version 探测略过 (lazy import; 不强制 hash, audio model 是 user 接受协议下载的)
    )
    _emit_event(state_dir, "diarize", "started", sidecar=current_sidecar)
    
    try:
        turns = diarize_audio(audio_path, hf_token=hf_token, allow_long=args.allow_long)
        payload = {"version": 1, "turns": turns}
        write_json_atomic(out_path, payload, sidecar_params=current_sidecar)
        _emit_event(state_dir, "diarize", "completed", sidecar=current_sidecar,
                    details={"turns_count": len(turns), "duration_s": duration_s})
        print(f"diarization: {len(turns)} turns")
        speakers = sorted({t["speaker_id"] for t in turns})
        print(f"speakers: {len(speakers)} unique ({', '.join(speakers[:5])}{'...' if len(speakers) > 5 else ''})")
        print(f"output: {out_path}")
    except Exception as e:
        _emit_event(state_dir, "diarize", "failed", sidecar=current_sidecar,
                    details={"error_type": type(e).__name__, "error": str(e)[:200]})
        raise

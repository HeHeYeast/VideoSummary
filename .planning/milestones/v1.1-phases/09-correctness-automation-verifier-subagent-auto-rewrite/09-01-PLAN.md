```python
"""Phase 09 CORR-03a — mechanical format-spec + citation + glossary checker.

K5 boundary: this module READS the summary artifact and produces a sibling
lint report; it NEVER modifies the summary artifact. The literal substring
"summary.md" appears in this file ONLY as (a) the LINT_FILENAME constant's
sibling-naming context comment and (b) the input filename in the function
signature `lint_summary(summary_path, ...)` — both are READ paths.

Statically asserted by tests/test_k5_emitters.py via _WRITE_PATTERNS_FORBIDDEN
regex (mirrors the Phase 08-01 agent/glossary.py exception: substring is
legitimate; what's forbidden is the WRITE pattern targeting the summary
artifact via write_text / open(...,'w') / os.replace / _atomic_write).
"""
from __future__ import annotations

import re
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path

LINT_FILENAME = "summary_lint.json"

# ── Format invariant patterns (locked at CLAUDE.md `### 格式锁定` 4+1 项) ──
# Invariant 1: timestamps are EXACTLY [HH:MM:SS] = 8 chars between brackets,
# with HH 00-99 / MM 00-59 / SS 00-59. We surface ANY [...] that LOOKS like
# a timestamp (digits + colons) but does NOT match the strict 8-char form.
_VALID_TIMESTAMP_RE = re.compile(r"\[(\d{2}):(\d{2}):(\d{2})\]")
_LOOSE_TIMESTAMP_RE = re.compile(r"\[(\d{1,2}(?::\d{1,2}){1,2})\]")  # catches violations

# Invariant 2: every fenced code block must declare a language.
# A bare ``` followed by a newline (no lang token) is a violation.
_CODE_FENCE_RE = re.compile(r"^```(\S*)\s*$")  # captures lang token; empty = violation

# Invariant 3: relative `frames/` paths only — disallow http://, https://,
# absolute paths starting with / or DRIVE: (Windows), or `..\frames`.
_FRAME_LINK_RE = re.compile(r"!\[[^\]]*\]\(([^)]+)\)")
_VALID_FRAME_PATH_RE = re.compile(r"^frames/[A-Za-z0-9_\-./]+\.(jpg|jpeg|png|webp)$")

# Invariant 4: 第二人称 imperative — heuristic: forbid "我们 ... <action>" and
# "<noun> 被 <action>" where action ∈ {打开,创建,配置,运行,执行,导入,启动}.
_FORBIDDEN_FIRST_PERSON_RE = re.compile(r"(我们|让我们)[^。\n]{0,8}?(打开|创建|配置|运行|执行|导入|启动)")
_FORBIDDEN_PASSIVE_RE = re.compile(r"\S{1,8}\s*被\s*(打开|创建|配置|运行|执行|导入|启动)")

# Invariant 5 (Phase 08): every load-bearing claim line must end with a
# trace token. We define a "load-bearing claim" line as a line that:
#   - is inside the body section (NOT in TL;DR / prelude / glossary)
#   - contains at least one of: a number, a fenced span `code`, a path,
#     a parameter-looking token (e.g. `fps=0.3`), or a UI control reference.
# The trace token must match either:
#   [seg_NNNN_NNNNNN.jpg @ HH:MM:SS]  OR  [para_NNNN @ HH:MM:SS]
_TRACE_TOKEN_RE = re.compile(
    r"\[(?:seg_\d{4}_\d{6}\.jpg|para_\d{4})\s*@\s*\d{2}:\d{2}:\d{2}\]"
)
_LOAD_BEARING_HINT_RE = re.compile(
    r"(?:`[^`]+`|\d+(?:\.\d+)?(?:px|s|fps|%|x)?|fps\s*=|\bfps\b|frames/|\.(json|md|py|sh|yml|yaml)|`[^`]+\.(?:json|md|py)`)"
)

# Sections in which trace tokens are FORBIDDEN (P-02 citation pollution rules,
# CLAUDE.md CORR-02 引用资格规则 → FORBIDDEN list)
_FORBIDDEN_CITATION_SECTIONS = {
    "5 分钟速读版": "TL;DR",
    "读这篇前你需要知道": "你需要知道什么",
    "你不需要知道什么": "你不需要知道什么",
    "章节小结": "transition",
    "总评": "transition",
}

_GLOSSARY_INLINE_RE = re.compile(r"\b([A-Z][A-Za-z0-9_-]{1,30})\s*\(([^)]{1,80})\)")  # term + paren definition
_GLOSSARY_H2_RE = re.compile(r"^##\s+([A-Z][A-Za-z0-9_-]{1,30})\s*\(([^)]{1,80})\)")


def _now_iso() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z")


def _classify_section(line: str, current_section: str | None) -> str | None:
    """Track the current H2 section as we walk the file line-by-line."""
    if line.startswith("## "):
        heading = line[3:].strip()
        for substr, label in _FORBIDDEN_CITATION_SECTIONS.items():
            if substr in heading:
                return label
        return "body"
    return current_section


def _check_timestamp_format(lines: list[str]) -> list[dict]:
    """Invariant 1: any `[N:NN]` / `[NN:N]` / `[1:23:45]` is a violation."""
    violations: list[dict] = []
    for i, line in enumerate(lines, start=1):
        for m in _LOOSE_TIMESTAMP_RE.finditer(line):
            full = f"[{m.group(1)}]"
            # Re-check against strict 8-char form: must be HH:MM:SS with each part 2 digits
            if not _VALID_TIMESTAMP_RE.match(full):
                violations.append({"line": i, "snippet": line.strip()[:120], "found": full})
    return violations


def _check_code_fence_language(lines: list[str]) -> list[dict]:
    """Invariant 2: every ``` opening fence must have a lang token."""
    violations: list[dict] = []
    in_fence = False
    for i, line in enumerate(lines, start=1):
        m = _CODE_FENCE_RE.match(line)
        if m:
            if not in_fence:
                lang = m.group(1)
                if not lang:
                    violations.append({"line": i, "snippet": line.strip()[:120]})
            in_fence = not in_fence
    return violations


def _check_relative_frame_paths(lines: list[str]) -> list[dict]:
    """Invariant 3: image src must be `frames/...jpg` (or .jpeg/.png/.webp)."""
    violations: list[dict] = []
    for i, line in enumerate(lines, start=1):
        for m in _FRAME_LINK_RE.finditer(line):
            src = m.group(1).strip()
            if not _VALID_FRAME_PATH_RE.match(src):
                violations.append({"line": i, "snippet": line.strip()[:120], "found": src})
    return violations


def _check_second_person_imperative(lines: list[str]) -> list[dict]:
    """Invariant 4: forbid 我们/passive constructions for action verbs."""
    violations: list[dict] = []
    for i, line in enumerate(lines, start=1):
        if _FORBIDDEN_FIRST_PERSON_RE.search(line) or _FORBIDDEN_PASSIVE_RE.search(line):
            violations.append({"line": i, "snippet": line.strip()[:120]})
    return violations


def _is_load_bearing(line: str) -> bool:
    return bool(_LOAD_BEARING_HINT_RE.search(line))


def _has_trace_token(line: str) -> bool:
    return bool(_TRACE_TOKEN_RE.search(line))


def _check_trace_after_claim(lines: list[str], section_map: list[str | None]) -> list[dict]:
    """Invariant 5 (Phase 08): load-bearing claim lines in body must end with trace token."""
    violations: list[dict] = []
    for i, line in enumerate(lines, start=1):
        section = section_map[i - 1]
        if section != "body":
            continue
        if _is_load_bearing(line) and not _has_trace_token(line):
            violations.append({"line": i, "snippet": line.strip()[:120]})
    return violations


def _compute_citation_stats(lines: list[str], section_map: list[str | None]) -> dict:
    claims_total = 0
    claims_with_trace = 0
    claims_without_trace: list[dict] = []
    uncertainty = 0
    for i, line in enumerate(lines, start=1):
        section = section_map[i - 1]
        if section == "body" and _is_load_bearing(line):
            claims_total += 1
            if _has_trace_token(line):
                claims_with_trace += 1
            else:
                claims_without_trace.append({"line": i, "snippet": line.strip()[:120]})
        uncertainty += line.count("[?]")
    density = (claims_with_trace / claims_total) if claims_total else 0.0
    return {
        "claims_total": claims_total,
        "claims_with_trace": claims_with_trace,
        "claims_without_trace": claims_without_trace,
        "trace_density": round(density, 3),
        "uncertainty_markers": uncertainty,
    }


def _check_citation_eligibility(lines: list[str], section_map: list[str | None]) -> list[dict]:
    """P-02 citation pollution: trace tokens are FORBIDDEN in TL;DR / prelude / transitions."""
    violations: list[dict] = []
    for i, line in enumerate(lines, start=1):
        section = section_map[i - 1]
        if section in ("TL;DR", "你需要知道什么", "你不需要知道什么", "transition") and _has_trace_token(line):
            violations.append({
                "line": i,
                "section": section,
                "reason": f"FORBIDDEN — {section} section contains trace token",
                "snippet": line.strip()[:120],
            })
    return violations


def _check_glossary_consistency(lines: list[str], glossary_path: Path | None) -> list[dict]:
    if glossary_path is None or not glossary_path.exists():
        return []
    glossary_defs: dict[str, str] = {}
    for gline in glossary_path.read_text(encoding="utf-8").splitlines():
        m = _GLOSSARY_H2_RE.match(gline)
        if m:
            term = m.group(1)
            definition = m.group(2).strip()
            glossary_defs.setdefault(term, definition)  # first-seen-wins
    drifts: list[dict] = []
    for line in lines:
        for m in _GLOSSARY_INLINE_RE.finditer(line):
            term = m.group(1)
            inline_def = m.group(2).strip()
            if term in glossary_defs and inline_def != glossary_defs[term]:
                drifts.append({
                    "term": term,
                    "summary_definition": inline_def,
                    "glossary_definition": glossary_defs[term],
                    "drift_detected": True,
                })
    return drifts


def lint_summary(summary_path: Path, glossary_path: Path | None = None) -> dict:
    """Mechanical lint of the summary artifact. Returns the schema-locked report dict."""

    lines = text.splitlines() or [""]
    section_map: list[str | None] = []
    current: str | None = None
    for line in lines:
        current = _classify_section(line, current)

    return {

        "schema_version": 1,
        "summary_path": str(summary_path),
        "checked_at": _now_iso(),
        "format_invariants": {

        },

        "citation_eligibility_violations": _check_citation_eligibility(lines, section_map),
